#include<iostream>
using namespace std;
char Postfix[30], Prefix[30];
struct node
{
char data;
node *next;
};
//============================================================
class stk
{
public:
int newtop=-1;
 int arr1[30];
int newempty()
{
if(newtop==-1)
return 1;
else
return 0;
}
int newfull()
{
if(newtop==30)
return 1;
else
return 0;
}
void newpush(int data)
{
if(!newfull())
{
newtop++;
arr1[newtop]=data;
}
else
{
cout<<"\n Stack is full!!!";
}
}
int newpop()
{
if(!newempty())
{
newtop--;
 return arr1[newtop+1];
}
else
{
cout<<"\n Stack is empty!!!";
}
return 0;
}
};
//============================================================
class Stack : public stk
{
public:
node *top;
Stack()
{
top=NULL;
}
int empty()
{
if(top==NULL)
{
return 1;
}
else
{
return 0;
}
}
void push(char c)
{
node *newNode;
newNode=new node;
newNode->data=c;
newNode->next=top;
top=newNode;
}
char pop()
{
if(!empty())
{
node *temp;
temp=top;
top=top->next;
return temp->data;;
}
return 0;
}
//------------------------------------------------------
int check_Operator_or_Operand(char ch)
{
if(ch=='+' || ch=='-' || ch=='*' || ch=='/' || ch==')' || ch=='(')
{
return 1;
}
else if(ch>='A' && ch<='Z' || ch>='a' && ch<='z')
{
return 0;
}
else
{
cout<<"\nWrong Expression !!!!!!!!";
}
return 0;
}
int precedance(char ch)
{
if(ch=='*' || ch=='/')
{
return 2;
}
else if(ch=='+' || ch=='-')
{
return 1;
}
return 0;
}
// ------------------------------------------------------------
void Infix_To_Postfix(char Infix[])
{
int i=0;
int j=0;
while(Infix[i]!='\0')
{
if(check_Operator_or_Operand(Infix[i])==0)
{
Postfix[j]=Infix[i];
j++;
}
else
{
if(top!=NULL && top->data!='(' && top->data!=')' &&
precedance(top->data)>=precedance(Infix[i]))
{
Postfix[j]=pop();
j++;
push(Infix[i]);
}
else
{
if(Infix[i]==')')
{
while(top->data!='(')
{
Postfix[j]=pop();
j++;
}
pop();
}
else
{
push(Infix[i]);
}
}
}
i++;
}
while(top!=NULL)
{
Postfix[j]=top->data;
 pop();
j++;
}
}

void evaluatePostfix()
{

{

}

// ------------------------------------------------------------
int evaluate(char x,int op1,int op2)
{
if(x=='+')
return(op1+op2);
if(x=='-')
return(op1-op2);
if(x=='*')
return(op1*op2);
if(x=='/')
return(op1/op2);
if(x=='%')
return(op1%op2);
return 0;
}

#include<iostream>
using namespace std;
int i=0,leafCount=0;
struct node
{
int data;
node *left;
node *right;
};
class BST
{
node *root;
public:
node* Create(int arr[], int n)
{
node *newNode,*temp,*ptr;
for(int i=0;i<n;i++)
{
newNode=new node;
newNode->data=arr[i];
newNode->left=NULL;
newNode->right=NULL;
if(i==0)
{
root=newNode;
}
else
{
temp=root;
while(temp!=NULL)
{
ptr=temp;
if(newNode->data<temp->data)
{
temp=temp->left;
}
else
{
temp=temp->right;
}
}
if(newNode->data<ptr->data)
{
ptr->left=newNode;
}
else
{
ptr->right=newNode;
}
}
}
return root;
}
void Display1(node *root)
{
if(root!=NULL)
{
cout<<" "<<root->data;
Display1(root->left);
Display1(root->right);
}
}
void Display2(node *root)
{
if(root!=NULL)
{
Display2(root->left);
cout<<" "<<root->data;
Display2(root->right);
}
}
void Display3(node *root)
{
if(root!=NULL)
{
Display3(root->left);
Display3(root->right);
cout<<" "<<root->data;
}
}
int Search(node *root,int key)
{
if(root!=NULL)
{
if(root->data==key)
{
cout<<"\nData found!!! ";
i=1;
}
Search(root->left,key);
Search(root->right,key);
}
return i;
}
int TreeDepth(node *root)
{
int hl,hr;
if(root==NULL)
{
return 0;
}
if(root->left==NULL && root->right==NULL)
{
return 1;
}
hl=TreeDepth(root->left);
hr=TreeDepth(root->right);
return (max(hl,hr)+1);
}
int DisplayLeaf(node *root)
{
if(root==NULL)
{
return 0;
}
if(root->left==NULL && root->right==NULL)
{
cout<<" "<<root->data;
leafCount++;
}
DisplayLeaf(root->left);
DisplayLeaf(root->right);
return 0;
}
};

{
node *ROOT;
int n;
cout<<"Enter how many node do you want to create :";

if(arr[j]!=temp)
{
arr[i]=temp;

else
{

}
}
if(i==0)
{
arr[i]=temp;
}
}
BST bst;
 int ch,b;

do

{

}
else if(ch==3)

else
{
return 0;
